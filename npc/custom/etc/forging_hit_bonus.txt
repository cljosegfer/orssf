//===== rAthena Script =======================================
//= Blacksmith Forging Hit Bonus
//===== Description: =========================================
//= Grants +2 Hit per skill level in each Blacksmith forging
//= skill (Dagger/Sword/Two-Handed Sword/Knuckle/Spear/Axe/
//= Mace Research), regardless of equipped weapon. Also grants
//= +2 Atk and +2 Hit per skill level of Weapon Research.
//= Recalculated on login and map change so skill resets take
//= effect without requiring a relog.
//============================================================
-	script	Forging Hit Bonus	-1,{
OnPCLoginEvent:
OnPCLoadMapEvent:
	.@hit = (getskilllv("BS_DAGGER") + getskilllv("BS_SWORD") + getskilllv("BS_TWOHANDSWORD") + getskilllv("BS_KNUCKLE") + getskilllv("BS_SPEAR") + getskilllv("BS_AXE") + getskilllv("BS_MACE")) * 3;
	.@research = getskilllv("BS_WEAPONRESEARCH");
	.@atk = .@research * 2;
	.@hit += .@research * 2;
	if (.@hit > 0 || .@atk > 0)
		bonus_script "{ bonus bHit," + .@hit + "; bonus bBaseAtk," + .@atk + "; }",86400,1032;
	end;
}
