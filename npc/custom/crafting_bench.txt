//===== rAthena Script =======================================
//= PoE Style Crafting Bench
//===== By: ==================================================
//= claudio
//===== Description: =========================================
//= Uses an "Exalted Orb" to add a random enchant
//= to an empty Random Option slot on equipped gear.
//= Uses a "Scour Orb" to clear all random options.
//============================================================

alberta,122,57,3	script	Belton	2417,{
	disable_items;

	mes "[Crafting Bench]";
	mes "This bench lets you modify random properties on your equipped gear.";
	mes " ";
	mes "^0055FF[Magic]^000000 1st-2nd mod: costs 1x " + getitemname(.orb_id);
	mes "^FF8800[Rare]^000000  3rd-4th mod: costs 1x " + getitemname(.steel_orb_id);
	mes "^FF0000[Scour]^000000 Remove all mods: costs 1x " + getitemname(.reset_orb_id);
	mes "^AA00AA[Annul]^000000  Remove last mod: costs 1x " + getitemname(.annul_orb_id);
	mes "^008800[Fossil]^000000  Add a restricted mod: costs 1x fossil item";
	next;

	switch (select("Exalt Equipment:Use Fossil:Annul Equipment:Scour Equipment:Cancel")) {
	case 1: .@op = 1; goto L_SlotMenu;
	case 2: .@op = 4; goto L_FossilMenu;
	case 3: .@op = 2; goto L_SlotMenu;
	case 4: .@op = 3; goto L_SlotMenu;
	case 5: close;
	}
	end;

L_FossilMenu:
	.@menu$ = ""; .@n = 0;
	if (countitem(.stat_fossil_id)     > 0) { .@menu$ += "Stat Fossil ("      + countitem(.stat_fossil_id)     + "x):"; .@fmap[.@n] = 0; .@n++; }
	if (countitem(.size_fossil_id)     > 0) { .@menu$ += "Size Fossil ("      + countitem(.size_fossil_id)     + "x):"; .@fmap[.@n] = 1; .@n++; }
	if (countitem(.fire_fossil_id)     > 0) { .@menu$ += "Fire Fossil ("      + countitem(.fire_fossil_id)     + "x):"; .@fmap[.@n] = 2; .@n++; }
	if (countitem(.water_fossil_id)    > 0) { .@menu$ += "Water Fossil ("     + countitem(.water_fossil_id)    + "x):"; .@fmap[.@n] = 3; .@n++; }
	if (countitem(.wind_fossil_id)     > 0) { .@menu$ += "Wind Fossil ("      + countitem(.wind_fossil_id)     + "x):"; .@fmap[.@n] = 4; .@n++; }
	if (countitem(.earth_fossil_id)    > 0) { .@menu$ += "Earth Fossil ("     + countitem(.earth_fossil_id)    + "x):"; .@fmap[.@n] = 5; .@n++; }
	if (countitem(.otherele_fossil_id) > 0) { .@menu$ += "Mixed Ele Fossil (" + countitem(.otherele_fossil_id) + "x):"; .@fmap[.@n] = 6; .@n++; }
	if (!.@n) { mes "[Crafting Bench]"; mes "You have no fossils."; close; }
	.@menu$ += "Cancel";
	.@pick = select(.@menu$) - 1;
	if (.@pick >= .@n) close;
	.@fossil_type = .@fmap[.@pick];
	.@fossil_allow_weapon = (.@fossil_type <= 1);
	.@fossil_allow_misc   = (.@fossil_type == 0);
	goto L_SlotMenu;

L_SlotMenu:
	.@menu$ = ""; .@n = 0;
	if (getequipisequiped(EQI_HAND_R) && ((.@op != 1 && .@op != 4) || getequiprandomoption(EQI_HAND_R, 3, ROA_ID) == 0) && (.@op == 1 || .@op == 4 || getequiprandomoption(EQI_HAND_R, 0, ROA_ID) > 0) && (.@op != 4 || .@fossil_allow_weapon))
		{ .@menu$ += "Right Hand (Weapon):"; .@smap[.@n] = EQI_HAND_R; .@n++; }
	if (getequipisequiped(EQI_ARMOR) && ((.@op != 1 && .@op != 4) || getequiprandomoption(EQI_ARMOR, 3, ROA_ID) == 0) && (.@op == 1 || .@op == 4 || getequiprandomoption(EQI_ARMOR, 0, ROA_ID) > 0))
		{ .@menu$ += "Body Armor:"; .@smap[.@n] = EQI_ARMOR; .@n++; }
	if (getequipisequiped(EQI_HAND_L) && ((.@op != 1 && .@op != 4) || getequiprandomoption(EQI_HAND_L, 3, ROA_ID) == 0) && (.@op == 1 || .@op == 4 || getequiprandomoption(EQI_HAND_L, 0, ROA_ID) > 0))
		{ .@menu$ += "Left Hand (Shield):"; .@smap[.@n] = EQI_HAND_L; .@n++; }
	if (getequipisequiped(EQI_GARMENT) && ((.@op != 1 && .@op != 4) || getequiprandomoption(EQI_GARMENT, 3, ROA_ID) == 0) && (.@op == 1 || .@op == 4 || getequiprandomoption(EQI_GARMENT, 0, ROA_ID) > 0))
		{ .@menu$ += "Garment:"; .@smap[.@n] = EQI_GARMENT; .@n++; }
	if (getequipisequiped(EQI_SHOES) && ((.@op != 1 && .@op != 4) || getequiprandomoption(EQI_SHOES, 3, ROA_ID) == 0) && (.@op == 1 || .@op == 4 || getequiprandomoption(EQI_SHOES, 0, ROA_ID) > 0))
		{ .@menu$ += "Shoes:"; .@smap[.@n] = EQI_SHOES; .@n++; }
	if (getequipisequiped(EQI_HEAD_TOP) && ((.@op != 1 && .@op != 4) || getequiprandomoption(EQI_HEAD_TOP, 3, ROA_ID) == 0) && (.@op == 1 || .@op == 4 || getequiprandomoption(EQI_HEAD_TOP, 0, ROA_ID) > 0) && (.@op != 4 || .@fossil_allow_misc))
		{ .@menu$ += "Upper Headgear:"; .@smap[.@n] = EQI_HEAD_TOP; .@n++; }
	if (getequipisequiped(EQI_HEAD_LOW) && ((.@op != 1 && .@op != 4) || getequiprandomoption(EQI_HEAD_LOW, 3, ROA_ID) == 0) && (.@op == 1 || .@op == 4 || getequiprandomoption(EQI_HEAD_LOW, 0, ROA_ID) > 0) && (.@op != 4 || .@fossil_allow_misc))
		{ .@menu$ += "Lower Headgear:"; .@smap[.@n] = EQI_HEAD_LOW; .@n++; }
	if (getequipisequiped(EQI_ACC_L) && ((.@op != 1 && .@op != 4) || getequiprandomoption(EQI_ACC_L, 3, ROA_ID) == 0) && (.@op == 1 || .@op == 4 || getequiprandomoption(EQI_ACC_L, 0, ROA_ID) > 0) && (.@op != 4 || .@fossil_allow_misc))
		{ .@menu$ += "Left Accessory:"; .@smap[.@n] = EQI_ACC_L; .@n++; }
	if (getequipisequiped(EQI_ACC_R) && ((.@op != 1 && .@op != 4) || getequiprandomoption(EQI_ACC_R, 3, ROA_ID) == 0) && (.@op == 1 || .@op == 4 || getequiprandomoption(EQI_ACC_R, 0, ROA_ID) > 0) && (.@op != 4 || .@fossil_allow_misc))
		{ .@menu$ += "Right Accessory:"; .@smap[.@n] = EQI_ACC_R; .@n++; }
	if (!.@n) {
		mes "[Crafting Bench]";
		if (.@op == 1 || .@op == 4) mes "No eligible equipment (all slots are full or incompatible).";
		else                         mes "No equipment has enchants to remove.";
		close;
	}
	.@menu$ += "Cancel";
	.@pick = select(.@menu$) - 1;
	if (.@pick >= .@n) close;
	if (.@op == 1)      callsub S_Exalt, .@smap[.@pick];
	else if (.@op == 2) callsub S_Annul, .@smap[.@pick];
	else if (.@op == 4) callsub S_Fossil, .@smap[.@pick], .@fossil_type;
	else                callsub S_Scour, .@smap[.@pick];
	end;

S_Exalt:
	.@slot = getarg(0);

	if (.@slot == EQI_HAND_R)       .@slot_name$ = "Right Hand (Weapon)";
	else if (.@slot == EQI_ARMOR)   .@slot_name$ = "Body Armor";
	else if (.@slot == EQI_HAND_L)  .@slot_name$ = "Left Hand (Shield)";
	else if (.@slot == EQI_GARMENT) .@slot_name$ = "Garment";
	else if (.@slot == EQI_SHOES)   .@slot_name$ = "Shoes";
	else if (.@slot == EQI_HEAD_TOP) .@slot_name$ = "Upper Headgear";
	else if (.@slot == EQI_HEAD_LOW) .@slot_name$ = "Lower Headgear";
	else if (.@slot == EQI_ACC_L)   .@slot_name$ = "Left Accessory";
	else if (.@slot == EQI_ACC_R)   .@slot_name$ = "Right Accessory";

	if (!getequipisequiped(.@slot)) {
		mes "[Crafting Bench]";
		mes "You have nothing equipped in the " + .@slot_name$ + " slot.";
		close;
	}

	// Find the first empty random option slot (max 4 mods)
	.@found_slot = -1;
	for (.@i = 0; .@i < 4; .@i++) {
		if (getequiprandomoption(.@slot, .@i, ROA_ID) == 0) {
			.@found_slot = .@i;
			break;
		}
	}
	if (.@found_slot == -1) {
		mes "[Crafting Bench]";
		mes "This item already has 4 random properties (maximum).";
		mes "Use a ^FF0000Scour Orb^000000 to reset it first.";
		close;
	}

	// Slots 0-1 = Magic tier (Iron); slots 2-3 = Rare tier (Steel)
	if (.@found_slot < 2) {
		.@cur_orb_id = .orb_id;
		.@tier_label$ = "^0055FF[Magic]^000000";
	} else {
		.@cur_orb_id = .steel_orb_id;
		.@tier_label$ = "^FF8800[Rare]^000000";
	}

	if (countitem(.@cur_orb_id) < 1) {
		mes "[Crafting Bench]";
		mes "You need 1x ^0000FF" + getitemname(.@cur_orb_id) + "^000000 to add a " + .@tier_label$ + " mod.";
		close;
	}

	// === Weighted enchant type selection ===
	// .ench_slot: 0=all, 1=weapon, 2=armor (body/shield/garment/shoes), 3=misc (headgear/accessory)
	.@is_weapon = (.@slot == EQI_HAND_R);
	.@is_misc   = (.@slot == EQI_HEAD_TOP || .@slot == EQI_HEAD_LOW || .@slot == EQI_ACC_L || .@slot == EQI_ACC_R);
	.@is_armor  = (!.@is_weapon && !.@is_misc);
	.@total_ench_weight = 0;
	for (.@e = 0; .@e < 24; .@e++) {
		.@ew[.@e] = 0;
		if (.ench_slot[.@e] == 0 ||
		    (.ench_slot[.@e] == 1 && .@is_weapon) ||
		    (.ench_slot[.@e] == 2 && .@is_armor)  ||
		    (.ench_slot[.@e] == 3 && .@is_misc)) {
			.@ew[.@e] = .ench_weight[.@e];
			.@total_ench_weight += .@ew[.@e];
		}
	}
	.@roll = rand(0, .@total_ench_weight - 1);
	.@cum = 0;
	.@ench = 0;
	for (.@e = 0; .@e < 24; .@e++) {
		.@cum += .@ew[.@e];
		if (.@roll < .@cum) { .@ench = .@e; break; }
	}

	// === Softmax tier selection: prob(tier_i) ∝ exp(-δᵢ/T), T=85 ===
	.@total_weight = 0;
	for (.@t = 0; .@t < 7; .@t++) {
		.@w[.@t] = 0;
		if (BaseLevel >= .tier_lvl[.@t]) {
			.@delta = BaseLevel - .tier_lvl[.@t];
			.@w[.@t] = .exp_table[.@delta];
			.@total_weight += .@w[.@t];
		}
	}
	.@roll = rand(0, .@total_weight - 1);
	.@cum = 0;
	.@tier = 0;
	for (.@t = 0; .@t < 7; .@t++) {
		.@cum += .@w[.@t];
		if (.@roll < .@cum) { .@tier = .@t; break; }
	}

	// === Value roll: min = prev tier max (or 1 for tier 0) ===
	if (.@tier == 0)
		.@val_min = 1;
	else
		.@val_min = .ev_max[.@ench * 7 + .@tier - 1];
	.@random_value = rand(.@val_min, .ev_max[.@ench * 7 + .@tier]);

	// === Resolve random option ID and display name ===
	if (.@ench == 0) {
		// VAR_<stat>AMOUNT: pick uniformly from STR/AGI/VIT/INT/DEX/LUK (IDs 3-8)
		.@random_stat_id = rand(3, 8);
		setarray .@sn$[0], "STR", "AGI", "VIT", "INT", "DEX", "LUK";
		.@opt_name$ = .@sn$[.@random_stat_id - 3];
	} else if (.@ench == 8) {
		// DAMAGE_SIZE_<size>_TARGET: pick uniformly Small/Medium/Large (IDs 157-159)
		.@random_stat_id = rand(157, 159);
		setarray .@sn$[0], "Small", "Medium", "Large";
		.@opt_name$ = "DMG vs " + .@sn$[.@random_stat_id - 157] + " Size";
	} else if (.@ench == 14) {
		// ATTR_TOLERACE_<ele>: pick uniformly Water..Undead (IDs 26-34)
		.@random_stat_id = rand(26, 34);
		setarray .@sn$[0], "Water", "Earth", "Fire", "Wind", "Poison", "Holy", "Dark", "Ghost", "Undead";
		.@opt_name$ = "Resist " + .@sn$[.@random_stat_id - 26];
	} else if (.@ench == 22) {
		// DAMAGE_SIZE_<size>_USER: Small/Medium/Large reduce size dmg taken (IDs 160-162)
		.@random_stat_id = rand(160, 162);
		setarray .@sn$[0], "Small", "Medium", "Large";
		.@opt_name$ = "Dmg from " + .@sn$[.@random_stat_id - 160] + " Size";
	} else {
		.@random_stat_id = .ench_opt[.@ench];
		switch (.@ench) {
		case 1:  .@opt_name$ = "ATK"; break;
		case 2:  .@opt_name$ = "MATK"; break;
		case 3:  .@opt_name$ = "ATK%"; break;
		case 4:  .@opt_name$ = "MATK%"; break;
		case 5:  .@opt_name$ = "ASPD%"; break;
		case 6:  .@opt_name$ = "DMG vs Normal"; break;
		case 7:  .@opt_name$ = "DMG vs Boss"; break;
		case 9:  .@opt_name$ = "Ranged DMG"; break;
		case 10: .@opt_name$ = "Max HP"; break;
		case 11: .@opt_name$ = "DEF"; break;
		case 12: .@opt_name$ = "Max SP"; break;
		case 13: .@opt_name$ = "MDEF"; break;
		case 15: .@opt_name$ = "Max HP%"; break;
		case 16: .@opt_name$ = "Max SP%"; break;
		case 17: .@opt_name$ = "Flee"; break;
		case 18: .@opt_name$ = "Cast Time"; break;
		case 19: .@opt_name$ = "After-cast Delay"; break;
		case 20: .@opt_name$ = "Dmg from Normal"; break;
		case 21: .@opt_name$ = "Dmg from Boss"; break;
		case 23: .@opt_name$ = "All Ele Resist"; break;
		}
	}

	mes "[Crafting Bench]";
	mes "This will consume 1x ^0000FF" + getitemname(.@cur_orb_id) + "^000000 and add a " + .@tier_label$ + " mod to your " + .@slot_name$ + ".";
	mes "^FF0000This cannot be undone without a Scour Orb.^000000";
	next;
	if (select("Exalt it!:Cancel") == 2) close;

	delitem .@cur_orb_id, 1;
	setrandomoption .@slot, .@found_slot, .@random_stat_id, .@random_value, 0;

	specialeffect2 EF_SUCCESS;
	mes "[Crafting Bench]";
	mes "Exalt successful! Mods on your " + .@slot_name$ + ":";
	for (.@i = 0; .@i < 4; .@i++) {
		.@mid  = getequiprandomoption(.@slot, .@i, ROA_ID);
		.@mval = getequiprandomoption(.@slot, .@i, ROA_VALUE);
		if (.@mid == 0) continue;
		switch (.@mid) {
		case 1:   .@mname$ = "Max HP"; break;
		case 2:   .@mname$ = "Max SP"; break;
		case 3:   .@mname$ = "STR"; break;
		case 4:   .@mname$ = "AGI"; break;
		case 5:   .@mname$ = "VIT"; break;
		case 6:   .@mname$ = "INT"; break;
		case 7:   .@mname$ = "DEX"; break;
		case 8:   .@mname$ = "LUK"; break;
		case 9:   .@mname$ = "Max HP%"; break;
		case 10:  .@mname$ = "Max SP%"; break;
		case 13:  .@mname$ = "ATK%"; break;
		case 14:  .@mname$ = "MATK%"; break;
		case 16:  .@mname$ = "ASPD%"; break;
		case 17:  .@mname$ = "ATK"; break;
		case 19:  .@mname$ = "MATK"; break;
		case 20:  .@mname$ = "DEF"; break;
		case 21:  .@mname$ = "MDEF"; break;
		case 23:  .@mname$ = "Flee"; break;
		case 25:  .@mname$ = "Neutral Resist"; break;
		case 26:  .@mname$ = "Water Resist"; break;
		case 27:  .@mname$ = "Earth Resist"; break;
		case 28:  .@mname$ = "Fire Resist"; break;
		case 29:  .@mname$ = "Wind Resist"; break;
		case 30:  .@mname$ = "Poison Resist"; break;
		case 31:  .@mname$ = "Holy Resist"; break;
		case 32:  .@mname$ = "Dark Resist"; break;
		case 33:  .@mname$ = "Ghost Resist"; break;
		case 34:  .@mname$ = "Undead Resist"; break;
		case 35:  .@mname$ = "All Ele Resist"; break;
		case 147: .@mname$ = "DMG vs Normal"; break;
		case 148: .@mname$ = "DMG vs Boss"; break;
		case 149: .@mname$ = "Dmg from Normal"; break;
		case 150: .@mname$ = "Dmg from Boss"; break;
		case 157: .@mname$ = "DMG vs Small Size"; break;
		case 158: .@mname$ = "DMG vs Medium Size"; break;
		case 159: .@mname$ = "DMG vs Large Size"; break;
		case 160: .@mname$ = "Dmg from Small Size"; break;
		case 161: .@mname$ = "Dmg from Medium Size"; break;
		case 162: .@mname$ = "Dmg from Large Size"; break;
		case 166: .@mname$ = "Ranged DMG"; break;
		case 170: .@mname$ = "Cast Time"; break;
		case 171: .@mname$ = "After-cast Delay"; break;
		default:  .@mname$ = "Unknown"; break;
		}
		if (.@i == .@found_slot)
			mes "^00FF00[NEW] +" + .@mval + " " + .@mname$ + "^000000";
		else
			mes "+" + .@mval + " " + .@mname$;
	}
	close;

S_Fossil:
	.@slot  = getarg(0);
	.@ftype = getarg(1);

	if      (.@slot == EQI_HAND_R)   .@slot_name$ = "Right Hand (Weapon)";
	else if (.@slot == EQI_ARMOR)    .@slot_name$ = "Body Armor";
	else if (.@slot == EQI_HAND_L)   .@slot_name$ = "Left Hand (Shield)";
	else if (.@slot == EQI_GARMENT)  .@slot_name$ = "Garment";
	else if (.@slot == EQI_SHOES)    .@slot_name$ = "Shoes";
	else if (.@slot == EQI_HEAD_TOP) .@slot_name$ = "Upper Headgear";
	else if (.@slot == EQI_HEAD_LOW) .@slot_name$ = "Lower Headgear";
	else if (.@slot == EQI_ACC_L)    .@slot_name$ = "Left Accessory";
	else if (.@slot == EQI_ACC_R)    .@slot_name$ = "Right Accessory";

	if (!getequipisequiped(.@slot)) {
		mes "[Crafting Bench]";
		mes "You have nothing equipped in the " + .@slot_name$ + " slot.";
		close;
	}

	.@found_slot = -1;
	for (.@i = 0; .@i < 4; .@i++) {
		if (getequiprandomoption(.@slot, .@i, ROA_ID) == 0) { .@found_slot = .@i; break; }
	}
	if (.@found_slot == -1) {
		mes "[Crafting Bench]";
		mes "This item already has 4 random properties (maximum).";
		mes "Use a ^FF0000Scour Orb^000000 to reset it first.";
		close;
	}

	switch (.@ftype) {
	case 0: .@fossil_item_id = .stat_fossil_id;     break;
	case 1: .@fossil_item_id = .size_fossil_id;     break;
	case 2: .@fossil_item_id = .fire_fossil_id;     break;
	case 3: .@fossil_item_id = .water_fossil_id;    break;
	case 4: .@fossil_item_id = .wind_fossil_id;     break;
	case 5: .@fossil_item_id = .earth_fossil_id;    break;
	case 6: .@fossil_item_id = .otherele_fossil_id; break;
	}
	if (countitem(.@fossil_item_id) < 1) {
		mes "[Crafting Bench]";
		mes "You need 1x ^0000FF" + getitemname(.@fossil_item_id) + "^000000 to use this fossil.";
		close;
	}

	// Resolve enchant index, option ID, and display name
	if (.@ftype == 0) {
		.@ench = 0;
		.@random_stat_id = rand(3, 8);
		setarray .@sn$[0], "STR", "AGI", "VIT", "INT", "DEX", "LUK";
		.@opt_name$ = .@sn$[.@random_stat_id - 3];
	} else if (.@ftype == 1) {
		if (.@slot == EQI_HAND_R) {
			.@ench = 8;
			.@random_stat_id = rand(157, 159);
			setarray .@sn$[0], "Small", "Medium", "Large";
			.@opt_name$ = "DMG vs " + .@sn$[.@random_stat_id - 157] + " Size";
		} else {
			.@ench = 22;
			.@random_stat_id = rand(160, 162);
			setarray .@sn$[0], "Small", "Medium", "Large";
			.@opt_name$ = "Dmg from " + .@sn$[.@random_stat_id - 160] + " Size";
		}
	} else if (.@ftype == 6) {
		.@ench = 14;
		.@idx = rand(0, 5);
		setarray .@rids[0],    25,  30,  31,  32,  33,  34;
		setarray .@rnames$[0], "Neutral Resist", "Poison Resist", "Holy Resist", "Dark Resist", "Ghost Resist", "Undead Resist";
		.@random_stat_id = .@rids[.@idx];
		.@opt_name$ = .@rnames$[.@idx];
	} else {
		.@ench = 14;
		switch (.@ftype) {
		case 2: .@random_stat_id = 28; .@opt_name$ = "Fire Resist";  break;
		case 3: .@random_stat_id = 26; .@opt_name$ = "Water Resist"; break;
		case 4: .@random_stat_id = 29; .@opt_name$ = "Wind Resist";  break;
		case 5: .@random_stat_id = 27; .@opt_name$ = "Earth Resist"; break;
		}
	}

	// Softmax tier selection
	.@total_weight = 0;
	for (.@t = 0; .@t < 7; .@t++) {
		.@w[.@t] = 0;
		if (BaseLevel >= .tier_lvl[.@t]) {
			.@delta = BaseLevel - .tier_lvl[.@t];
			.@w[.@t] = .exp_table[.@delta];
			.@total_weight += .@w[.@t];
		}
	}
	.@roll = rand(0, .@total_weight - 1); .@cum = 0; .@tier = 0;
	for (.@t = 0; .@t < 7; .@t++) {
		.@cum += .@w[.@t];
		if (.@roll < .@cum) { .@tier = .@t; break; }
	}

	if (.@tier == 0)
		.@val_min = 1;
	else
		.@val_min = .ev_max[.@ench * 7 + .@tier - 1];
	.@random_value = rand(.@val_min, .ev_max[.@ench * 7 + .@tier]);

	mes "[Crafting Bench]";

	mes "^FF0000This cannot be undone without a Scour Orb.^000000";

S_Scour:
	.@slot = getarg(0);

	if (.@slot == EQI_HAND_R)       .@slot_name$ = "Right Hand (Weapon)";
	else if (.@slot == EQI_ARMOR)   .@slot_name$ = "Body Armor";
	else if (.@slot == EQI_HAND_L)  .@slot_name$ = "Left Hand (Shield)";
	else if (.@slot == EQI_GARMENT) .@slot_name$ = "Garment";
	else if (.@slot == EQI_SHOES)   .@slot_name$ = "Shoes";
	else if (.@slot == EQI_HEAD_TOP) .@slot_name$ = "Upper Headgear";
	else if (.@slot == EQI_HEAD_LOW) .@slot_name$ = "Lower Headgear";
	else if (.@slot == EQI_ACC_L)   .@slot_name$ = "Left Accessory";
	else if (.@slot == EQI_ACC_R)   .@slot_name$ = "Right Accessory";

	if (!getequipisequiped(.@slot)) {
		mes "[Crafting Bench]";
		mes "You have nothing equipped in the " + .@slot_name$ + " slot.";
		close;
	}

	.@has_options = 0;
	for (.@i = 0; .@i < 5; .@i++) {
		if (getequiprandomoption(.@slot, .@i, ROA_ID) > 0) {
			.@has_options = 1;
			break;
		}
	}
	if (!.@has_options) {
		mes "[Crafting Bench]";
		mes "This item has no random properties to remove.";
		close;
	}
	if (countitem(.reset_orb_id) < 1) {
		mes "[Crafting Bench]";
		mes "You need 1x ^0000FF" + getitemname(.reset_orb_id) + "^000000 to Scour.";
		close;
	}

	// Save item state before unequipping
	.@equip_id = getequipid(.@slot);
	.@equip_refine = getequiprefinerycnt(.@slot);
	setarray .@equip_card[0], getequipcardid(.@slot,0), getequipcardid(.@slot,1), getequipcardid(.@slot,2), getequipcardid(.@slot,3);

	mes "[Crafting Bench]";
	mes "This will consume 1x ^0000FF" + getitemname(.reset_orb_id) + "^000000 and remove ^FF0000ALL^000000 random properties from your " + .@slot_name$ + ".";
	next;
	if (select("Scour it!:Cancel") == 2) close;

	delitem .reset_orb_id, 1;
	delequip .@slot;
	getitem3 .@equip_id, 1, 1, .@equip_refine, 0, .@equip_card[0], .@equip_card[1], .@equip_card[2], .@equip_card[3], .@zid, .@zval, .@zparam;

	specialeffect2 EF_REPAIRWEAPON;
	mes "[Crafting Bench]";
	mes "All random properties have been removed from your " + .@slot_name$ + ".";
	close;

S_Annul:
	.@slot = getarg(0);

	if (.@slot == EQI_HAND_R)       .@slot_name$ = "Right Hand (Weapon)";
	else if (.@slot == EQI_ARMOR)   .@slot_name$ = "Body Armor";
	else if (.@slot == EQI_HAND_L)  .@slot_name$ = "Left Hand (Shield)";
	else if (.@slot == EQI_GARMENT) .@slot_name$ = "Garment";
	else if (.@slot == EQI_SHOES)   .@slot_name$ = "Shoes";
	else if (.@slot == EQI_HEAD_TOP) .@slot_name$ = "Upper Headgear";
	else if (.@slot == EQI_HEAD_LOW) .@slot_name$ = "Lower Headgear";
	else if (.@slot == EQI_ACC_L)   .@slot_name$ = "Left Accessory";
	else if (.@slot == EQI_ACC_R)   .@slot_name$ = "Right Accessory";

	if (!getequipisequiped(.@slot)) {
		mes "[Crafting Bench]";
		mes "You have nothing equipped in the " + .@slot_name$ + " slot.";
		close;
	}

	// Find the last filled random option slot (scan 3 → 0)
	.@last_slot = -1;
	for (.@i = 3; .@i >= 0; .@i--) {
		if (getequiprandomoption(.@slot, .@i, ROA_ID) > 0) {
			.@last_slot = .@i;
			break;
		}
	}
	if (.@last_slot == -1) {
		mes "[Crafting Bench]";
		mes "This item has no enchants to remove.";
		close;
	}

	if (countitem(.annul_orb_id) < 1) {
		mes "[Crafting Bench]";
		mes "You need 1x ^0000FF" + getitemname(.annul_orb_id) + "^000000 to Annul.";
		close;
	}

	// Read all 4 option slots
	for (.@i = 0; .@i < 4; .@i++) {
		.@opt_id[.@i]    = getequiprandomoption(.@slot, .@i, ROA_ID);
		.@opt_val[.@i]   = getequiprandomoption(.@slot, .@i, ROA_VALUE);
		.@opt_param[.@i] = getequiprandomoption(.@slot, .@i, ROA_PARAM);
	}

	// Zero out the last filled slot
	.@opt_id[.@last_slot]    = 0;
	.@opt_val[.@last_slot]   = 0;
	.@opt_param[.@last_slot] = 0;

	// Save item state
	.@equip_id = getequipid(.@slot);
	.@equip_refine = getequiprefinerycnt(.@slot);
	setarray .@equip_card[0], getequipcardid(.@slot,0), getequipcardid(.@slot,1), getequipcardid(.@slot,2), getequipcardid(.@slot,3);

	mes "[Crafting Bench]";
	mes "This will consume 1x ^0000FF" + getitemname(.annul_orb_id) + "^000000 and remove the ^FF0000last^000000 random property from your " + .@slot_name$ + ".";
	next;
	if (select("Annul it!:Cancel") == 2) close;

	delitem .annul_orb_id, 1;
	delequip .@slot;
	getitem3 .@equip_id, 1, 1, .@equip_refine, 0, .@equip_card[0], .@equip_card[1], .@equip_card[2], .@equip_card[3], .@opt_id, .@opt_val, .@opt_param;

	specialeffect2 EF_REPAIRWEAPON;
	mes "[Crafting Bench]";
	mes "Last enchant removed from your " + .@slot_name$ + ".";
	close;
